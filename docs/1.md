1. 핵심 인사이트
판다랭크(Pandarank)의 핵심 아키텍처는 블로그 페르소나 분석 → 실시간/황금 키워드 매칭 → 에디터 최적화 AI 포스팅 생성 → 숏폼/수익화 자동 확장으로 이어지는 파이프라인에 있습니다.

기존에 구축된 KeywordOn(keywordon.vercel.app)의 검색량·기회지수 엔진 위에, 판다랭크 내부 화면에서 확인된 '6축 블로그 육각형 진단', '블로그 페르소나/말투 복제 엔진', '원클릭 숏폼 대본/글감 추천기'를 통합하면 상용 유료 툴 이상의 개인 전용 인텔리전스 시스템을 완성할 수 있습니다.

[모듈 1] 블로그 글 URL 형태소 정밀 진단 API
경로: app/api/audit/post/route.ts

Copyimport { NextRequest, NextResponse } from 'next/server';
import * as cheerio from 'cheerio';

export const runtime = 'nodejs';

interface AuditResponse {
  title: string;
  charCountWithSpace: number;
  charCountWithoutSpace: number;
  imageCount: number;
  hasVideo: boolean;
  targetKeyword: string;
  keywordCount: number;
  keywordDensity: number;
  seoScore: number;
  diagnostics: {
    lengthStatus: '부족' | '적정' | '과다';
    densityStatus: '희박' | '최적' | '과다(어뷰징위험)';
    imageStatus: '부족' | '적정';
    overallMessage: string;
  };
}

export async function POST(req: NextRequest) {
  try {
    const { postUrl, targetKeyword } = await req.json();

    if (!postUrl) {
      return NextResponse.json({ error: '블로그 포스팅 URL을 입력해주세요.' }, { status: 400 });
    }

    // 네이버 블로그 스마트에디터 파싱 URL로 정규화
    let targetUrl = postUrl;
    if (postUrl.includes('blog.naver.com') && !postUrl.includes('PostView.naver')) {
      const match = postUrl.match(/blog\.naver\.com\/([a-zA-Z0-9_-]+)\/(\d+)/);
      if (match) {
        targetUrl = `https://blog.naver.com/PostView.naver?blogId=${match[1]}&logNo=${match[2]}&redirect=Dlog&widgetTypeCall=true`;
      }
    }

    const response = await fetch(targetUrl, {
      headers: {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
      },
    });

    if (!response.ok) {
      return NextResponse.json({ error: '포스팅 데이터를 불러올 수 없습니다.' }, { status: 404 });
    }

    const html = await response.text();
    const $ = cheerio.load(html);

    const title = $('.se-title-text').text().trim() || $('title').text().trim();
    const contentArea = $('.se-main-container').length > 0 ? $('.se-main-container') : $('#postViewArea');
    
    // 불필요한 스크립트 및 스타일 태그 제거
    contentArea.find('script, style').remove();
    const rawText = contentArea.text().replace(/\s+/g, ' ').trim();

    const charCountWithSpace = rawText.length;
    const charCountWithoutSpace = rawText.replace(/\s/g, '').length;
    const imageCount = contentArea.find('img').length;
    const hasVideo = contentArea.find('iframe, video, .se-video').length > 0;

    // 타깃 키워드 분석
    const kw = targetKeyword ? targetKeyword.trim() : '';
    let keywordCount = 0;
    let keywordDensity = 0;

    if (kw && charCountWithoutSpace > 0) {
      const regex = new RegExp(kw, 'gi');
      const matches = rawText.match(regex);
      keywordCount = matches ? matches.length : 0;
      keywordDensity = Number(((keywordCount * kw.length) / charCountWithoutSpace * 100).toFixed(2));
    }

    // 진단 지표 산출
    let lengthScore = 0;
    let lengthStatus: '부족' | '적정' | '과다' = '적정';
    if (charCountWithoutSpace < 1200) {
      lengthScore = (charCountWithoutSpace / 1200) * 30;
      lengthStatus = '부족';
    } else if (charCountWithoutSpace <= 3500) {
      lengthScore = 30;
      lengthStatus = '적정';
    } else {
      lengthScore = 25;
      lengthStatus = '과다';
    }

    let densityScore = 0;
    let densityStatus: '희박' | '최적' | '과다(어뷰징위험)' = '최적';
    if (keywordDensity < 1.0) {
      densityScore = 15;
      densityStatus = '희박';
    } else if (keywordDensity <= 3.0) {
      densityScore = 40;
      densityStatus = '최적';
    } else {
      densityScore = 10;
      densityStatus = '과다(어뷰징위험)';
    }

    const imageScore = Math.min(30, imageCount * 3.5);
    const imageStatus = imageCount >= 6 ? '적정' : '부족';
    const seoScore = Math.round(lengthScore + densityScore + imageScore);

    const result: AuditResponse = {
      title,
      charCountWithSpace,
      charCountWithoutSpace,
      imageCount,
      hasVideo,
      targetKeyword: kw,
      keywordCount,
      keywordDensity,
      seoScore,
      diagnostics: {
        lengthStatus,
        densityStatus,
        imageStatus,
        overallMessage: seoScore >= 80 ? '상위 노출에 최적화된 우수한 포스팅입니다.' : '글자수 또는 키워드 밀도 보완이 필요합니다.',
      },
    };

    return NextResponse.json({ success: true, data: result });
  } catch (error: any) {
    return NextResponse.json({ error: error.message || '분석 중 서버 오류가 발생했습니다.' }, { status: 500 });
  }
}

`;

  } catch (error: any) {

export const runtime = 'nodejs';

export async function POST(req: NextRequest) {
  try {
    const { postUrl } = await req.json();
    const apiKey = process.env.GEMINI_API_KEY;

    if (!postUrl || !apiKey) {
